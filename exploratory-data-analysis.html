Question 1

What is the total distribution of vote counts at the district level ?

Answer 1

district_level <-  sr_precinct |>
    filter(!(DEM02 > 0),
    !(REP02 > 0))|>
    group_by(DISTRICT)|>
    summarize(DEM = sum(DEM01, na.rm= T), 
    REP = sum(REP01, na.rm = T),
    TOTVOTE = sum(TOTVOTE, na.rm = T))|>
        mutate(WINNER= ifelse(DEM > REP, "DEM", "REP"))

district_level |>
    ggplot(aes(x = TOTVOTE))+geom_histogram(fill = "goldenrod")+ 
    labs(title = "Distribution of votes (district lvl)", 
    x = "#Total votes", 
    y  = "Count")+
    theme_minimal(base_family = "Arial")
`stat_bin()` using `bins = 30`. Pick better value `binwidth`.

Question 2 :

What is the association between number of democrat vote and number of republican vote at the district level ?

Answer 2:

m2 <- lm((REP)~(DEM), data = district_level)

g2 <-  district_level |>
    ggplot(aes(x = DEM, y = REP))+
    geom_smooth(method = "lm", color = "lightblue", se = F)+
    geom_point(aes(text = paste(
    "DISTRICT:", DISTRICT,
    "<br>Democrat votes:", DEM,
    "<br>Republican votes:", REP,
    "<br>Total votes:", TOTVOTE), color = WINNER))+
    scale_color_manual(
    values = c(
      "DEM" = "#6090F7",
      "REP" = "#FA6E52"))+ 
    theme_minimal(base_family = "Arial")+
    labs(title = "Association: dem and rep vote (district lvl)", 
    x = "# of Democrat votes", 
    y = "# of Republican votes", 
    caption = paste("the slope coefficient of our linear model is of :",round(m2[[1]][2], 4)))

ggplotly(g2)

Question 3:

What is the district with the highest number of republican votes given their number of democrat votes ?

Answer 3:

district_level |>
    ungroup()|>
    mutate(residual = resid(m2))|>
    arrange(desc(residual))|>
    select(DISTRICT)|>
    slice(1)
# A tibble: 1 × 1
  DISTRICT
     <int>
1        1

The 1st district has the highest number of republican vote given the number of democrat votes.

Question 5 :

What is the association between number of democrat vote and number of republican vote at the county level ?

Answer 5:

m1 <- lm((REP)~(DEM), data = county_level)

g1 <-  county_level |>
    ggplot(aes(x = DEM, y = REP))+
    geom_smooth(method = "lm", color = "lightblue", se = F)+
    geom_point(aes(text = paste(
    "COUNTY:", COUNTY,
    "<br>Democrat votes:", DEM,
    "<br>Republican votes:", REP,
    "<br>Total votes:", TOTVOTE), color = WINNER))+
    scale_color_manual(
    values = c(
      "DEM" = "#6090F7",
      "REP" = "#FA6E52"))+ 
    scale_y_log10()+
    scale_x_log10()+
    theme_minimal(base_family = "Arial")+
    labs(title = "Association: dem and rep vote (county lvl)", 
    x = "# of Democrat votes", 
    y = "# of Republican votes", 
    caption = paste("the slope coefficient of our linear model is of :",round(m1[[1]][2], 4)))

ggplotly(g1)
